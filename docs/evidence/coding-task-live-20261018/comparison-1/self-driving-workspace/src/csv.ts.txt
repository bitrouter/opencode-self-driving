export function parseCsv(text: string): string[][] {
  return text
    .replace(/^\uFEFF/, "")
    .split(/\r?\n/)
    .map((line) => {
      const fields = [];
      let field = '';
      let inQuotes = false;
      for (let i = 0; i < line.length; i++) {
        const char = line[i];
        if (inQuotes) {
          if (char === '"') {
            if (line[i + 1] === '"') {
              field += '"';
              i++;
            } else {
              inQuotes = false;
            }
          } else {
            field += char;
          }
        } else {
          if (char === '"' && field === '') {
            inQuotes = true;
          } else if (char === ',') {
            fields.push(field);
            field = '';
          } else {
            if (char === '\r') {
              throw new Error('Bare CR inside unquoted field');
            }
            field += char;
          }
        }
      }
      if (inQuotes) {
        throw new Error('Unterminated quoted field');
      }
      fields.push(field);
      return fields;
    });
}
